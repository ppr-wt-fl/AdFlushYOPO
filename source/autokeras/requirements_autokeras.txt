# Dependencies for train_autokeras.py (AutoKeras StructuredDataClassifier ->
# tf.js pipeline). Mirrors source/requirements_optuna.txt's pinned stack,
# which is already confirmed in this repo to produce a valid, trainable tfjs
# LayersModel.
#
# tensorflow==2.15.* is a deliberate pin: it's the last minor series where
# tf.keras is still Keras 2 by default (2.16+ defaults to Keras 3), which is
# the format tensorflowjs's save_keras_model() is best tested against.
#
# autokeras==1.1.0 is the last AutoKeras release built on tf.keras/Keras 2
# (autokeras>=2.0 requires keras>=3.0, which is incompatible with the
# Keras-2-only tfjs export path above -- see the Dockerfile header for the
# full investigation). 1.1.0 declares tensorflow>=2.8.0 with no upper pin,
# so it installs against TF 2.15 fine; nothing in its 1.1.0-era code touches
# TF APIs removed/changed since 2.8.
#
# autokeras unconditionally does `import keras_nlp` in its __init__.py (even
# though StructuredDataClassifier never touches NLP code), which pulls in
# keras-nlp. keras-nlp>=0.6 depends on `keras-core`, an early Keras-3
# precursor package that would fight the Keras-2 pin above, so keras-nlp is
# pinned to the last pre-keras-core release (0.5.x) instead. keras-nlp in
# turn needs tensorflow-text, pinned to match TF exactly.
#
# Every direct dependency is pinned to an exact version for reproducibility.
# For a fully hash-locked install, generate a lock from the first good build
# (`pip freeze > requirements_autokeras.lock` inside the built image, or
# `pip-compile --generate-hashes`) and point the Dockerfile at that instead.
numpy==1.26.4
tensorflow==2.15.0
tf-keras==2.15.1
tensorflowjs==4.22.0
autokeras==1.1.0
keras-nlp==0.5.2
# keras-nlp 0.5.2's BytePairTokenizer (imported transitively the moment
# `import autokeras` pulls in keras_nlp's package __init__, even though
# StructuredDataClassifier never uses NLP tokenizers) does `import regex`
# without declaring it in its own install_requires -- a packaging bug in
# that release. Installing it explicitly avoids a ModuleNotFoundError.
regex==2024.11.6
tensorflow-text==2.15.0
keras-tuner==1.4.7
scikit-learn==1.5.2
pandas==2.2.3
